# cp -av /usr/share/doc/dhcp-server/dhcpd.conf.example /etc/dhcp/dhcpd.conf

# vim /etc/dhcp/dhcpd.conf



option domain-name "sniper.com";
option domain-name-servers 192.168.159.100,192.168.159.200,192.168.159.2;




default-lease-time 600;
max-lease-time 7200;

authoritative;
allow bootp;

# A slightly different configuration for an internal subnet.
subnet 192.168.206.0 netmask 255.255.255.0 {
  range 192.168.206.110 192.168.206.200;
  option routers 192.168.206.2;
}


:wq
