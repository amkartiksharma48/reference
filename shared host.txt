Shared DNS: bts.army [ forward zone ]

# vim /etc/named.conf

Go to end of file and add

zone "bts.army" IN {
		type master;
		file "for.bts.army";
};

:wq

# cd /var/named
# vim for.bts.army
$TTL 1D
@       IN SOA  master.itiss.com. email@gmail.com. (
                                        0       ; serial
                                        1D      ; refresh
                                        1H      ; retry
                                        1W      ; expire
                                        3H )    ; minimum
                        IN      NS      master.itiss.com.
bts.army.               IN      A       192.168.206.151
www		        IN      A       192.168.206.151

:wq

# systemctl restart named

Step 3 — Managing the Apache Process
`````````````````````````````````````
[+] Enable user's HOME DIR as default DocumentRoot for apache web server:
````````````````````````````````````````````
# vim /etc/httpd/conf.d/userdir.conf

UserDir public_html

:wq 
___________________________________
/var/www/html	
Eg:
/home/user/public_html/index.html
/home/user1/public_html/index.html
____________________________________






# useradd user
# chmod 711 /home/user
# mkdir /home/user/public_html
# chmod 755 /home/user/public_html
# cat > /home/user/public_html/index.html
<h1>WEB SERVER</h1>
# chown user:user /home/user/public_html -R
# chmod 644 /home/user/public_html/index.html

Apache[httpd]: 
[x] 	1 	/home/user
[rx] 	5 	/home/user/public_html
[r]  	4 	/home/user/public_html/index.html










# mkdir /etc/httpd/sites-available /etc/httpd/sites-enabled

sites-available: directory to store Apache virtual host configuration files

sites-enabled  : directory will hold symbolic links/shortcut to virtual hosts, to enable the conf file


# vim /etc/httpd/conf/httpd.conf

Go to end of file and add

IncludeOptional sites-enabled/*.conf

:wq

# setenforce 0    # Disable SELinux 


# vim /etc/httpd/sites-available/sniper.conf

<VirtualHost *:80>
    ServerName sniper.com
    ServerAlias www.sniper.com
    DocumentRoot /var/www/html/sniper.com/
    ErrorLog /var/log/httpd/sniper_error_log
    CustomLog /var/log/httpd/sniper_access_log combined
</VirtualHost>
:wq


# ln -s /etc/httpd/sites-available/sniper.conf /etc/httpd/sites-enabled/sniper.conf

ln : Link
-s : Symbolic/Soft Link
