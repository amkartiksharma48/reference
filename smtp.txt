# SMTP SERVER CONFIGURATION — CENTOS STREAM 9

dnf install postfix -y

hostname -f
getent hosts mail.itiss.com

vim /etc/postfix/main.cf

Configure these lines in main.cf:

myhostname = mail.itiss.com
mydomain = itiss.com
myorigin = $mydomain
mydestination = $myhostname, localhost.$mydomain, localhost, $mydomain
inet_interfaces = all
mynetworks = 127.0.0.0/8, 192.168.178.0/24

Save and exit:
Esc
:wq

systemctl restart postfix
systemctl enable postfix
systemctl status postfix

firewall-cmd --permanent --add-service=smtp
firewall-cmd --reload

postconf -n
ss -tlnp | grep :25

useradd user1

useradd user2
passwd user2

# Send mail from user1 to user2
su - user1
echo "SMTP test mail" | mail -s "SMTP Test" user2

# Check mail as user2
su - user2
mail

# NOTE
# Replace mail.itiss.com, itiss.com and the mynetworks subnet
# with the actual values used in your lab.
# Local mail testing checks local delivery; it does not prove
# delivery to external email providers.
